-- 1. Métodos de pago
CREATE TABLE metodos_pagos (
    metodo_pago_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    descripcion_metodo VARCHAR2(50) NOT NULL
);

-- 2. Roles
CREATE TABLE rol (
    rol_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_rol VARCHAR2(30) NOT NULL,
    descripcion_rol VARCHAR2(100)
);

-- 3. Tipos de clientes
CREATE TABLE tipos_clientes (
    tipo_cliente_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    descripcion_tipo VARCHAR2(50) NOT NULL
);

-- 4. Nacionalidades
CREATE TABLE nacionalidades (
    nacionalidad_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_nacionalidad VARCHAR2(50) NOT NULL
);

-- 5. Países
CREATE TABLE paises (
    pais_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_pais VARCHAR2(50)
);

-- 6. Provincias
CREATE TABLE provincias (
    provincia_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    pais_id NUMBER,
    nombre_provincia VARCHAR2(50),
    FOREIGN KEY (pais_id) REFERENCES paises(pais_id)
);

-- 7. Distritos
CREATE TABLE distritos (
    distrito_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    provincia_id NUMBER,
    nombre_distrito VARCHAR2(50),
    FOREIGN KEY (provincia_id) REFERENCES provincias(provincia_id)
);

-- 8. Corregimientos
CREATE TABLE corregimientos (
    corregimiento_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    distrito_id NUMBER,
    nombre_corregimiento VARCHAR2(50),
    FOREIGN KEY (distrito_id) REFERENCES distritos(distrito_id)
);

-- 9. Direcciones
CREATE TABLE direcciones (
    direccion_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    corregimiento_id NUMBER,
    detalle_direccion VARCHAR2(200),
    FOREIGN KEY (corregimiento_id) REFERENCES corregimientos(corregimiento_id)
);

-- 10. Usuarios
CREATE TABLE usuarios (
    usuario_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_usuario VARCHAR2(50) NOT NULL,
    email_usuario VARCHAR2(100) NOT NULL,
    clave_hash VARCHAR2(256) NOT NULL,
    rol_id NUMBER NOT NULL,
    estado VARCHAR2(20) DEFAULT 'ACTIVO',
    fecha_creacion DATE DEFAULT SYSDATE,
    FOREIGN KEY (rol_id) REFERENCES rol(rol_id)
);

-- 11. Clientes
CREATE TABLE clientes (
    cliente_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    primer_nombre VARCHAR2(50) NOT NULL,
    segundo_nombre VARCHAR2(50),
    primer_apellido VARCHAR2(50) NOT NULL,
    segundo_apellido VARCHAR2(50),
    numero_cedula VARCHAR2(25) UNIQUE NOT NULL,
    telefono VARCHAR2(20),
    email VARCHAR2(100),
    sexo CHAR(1) CHECK (sexo IN ('M', 'F', 'O')),
    tipo_cliente_id NUMBER NOT NULL,
    direccion_id NUMBER,
    nacionalidad_id NUMBER,
    FOREIGN KEY (tipo_cliente_id) REFERENCES tipos_clientes(tipo_cliente_id),
    FOREIGN KEY (nacionalidad_id) REFERENCES nacionalidades(nacionalidad_id),
    FOREIGN KEY (direccion_id) REFERENCES direcciones(direccion_id)
);

-- 12. Sucursales
CREATE TABLE sucursales (
    sucursal_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_sucursal VARCHAR2(50),
    telefono VARCHAR2(20),
    direccion_id NUMBER,
    FOREIGN KEY (direccion_id) REFERENCES direcciones(direccion_id)
);

-- 13. Marcas de vehículos
CREATE TABLE marcas_vehiculos (
    marca_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_marca VARCHAR2(50) NOT NULL
);

-- 14. Modelos de vehículos
CREATE TABLE modelos_vehiculos (
    modelo_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    marca_id NUMBER NOT NULL,
    nombre_modelo VARCHAR2(50),
    anio_inicio NUMBER(4),
    FOREIGN KEY (marca_id) REFERENCES marcas_vehiculos(marca_id)
);

-- 15. Tipos de combustible
CREATE TABLE tipos_combustible (
    tipo_combustible_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    descripcion_combustible VARCHAR2(30)
);

-- 16. Vehículos
CREATE TABLE vehiculos (
    vehiculo_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    numero_placa VARCHAR2(15) UNIQUE,
    modelo_id NUMBER NOT NULL,
    tipo_combustible_id NUMBER NOT NULL,
    cliente_id NUMBER NOT NULL,
    color VARCHAR2(30),
    anio_fabricacion NUMBER(4),
    FOREIGN KEY (modelo_id) REFERENCES modelos_vehiculos(modelo_id),
    FOREIGN KEY (tipo_combustible_id) REFERENCES tipos_combustible(tipo_combustible_id),
    FOREIGN KEY (cliente_id) REFERENCES clientes(cliente_id)
);

-- 17. Productos
CREATE TABLE productos (
    producto_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    codigo_producto VARCHAR2(20) NOT NULL UNIQUE,
    nombre_producto VARCHAR2(100),
    cantidad_disponible NUMBER(10,2),
    precio_base NUMBER(10,2),
    precio_venta NUMBER(10,2)
);

-- 18. Inventario por sucursal
CREATE TABLE inventario_sucursal (
    inventario_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    producto_id NUMBER NOT NULL,
    sucursal_id NUMBER NOT NULL,
    cantidad_actual NUMBER(8,2),
    FOREIGN KEY (producto_id) REFERENCES productos(producto_id),
    FOREIGN KEY (sucursal_id) REFERENCES sucursales(sucursal_id)
);

-- 19. Personal
CREATE TABLE personal (
    personal_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    primer_nombre VARCHAR2(50) NOT NULL,
    segundo_nombre VARCHAR2(50),
    primer_apellido VARCHAR2(50) NOT NULL,
    segundo_apellido VARCHAR2(50),
    numero_cedula VARCHAR2(25) UNIQUE NOT NULL,
    telefono VARCHAR2(20),
    email VARCHAR2(100),
    sexo CHAR(1) CHECK (sexo IN ('M', 'F', 'O')),
    direccion_id NUMBER,
    usuario_id NUMBER,
    fecha_contratacion DATE DEFAULT SYSDATE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(usuario_id),
    FOREIGN KEY (direccion_id) REFERENCES direcciones(direccion_id)
);

-- 20. Servicios
CREATE TABLE servicios (
    servicio_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_servicio VARCHAR2(100),
    descripcion_servicio VARCHAR2(200),
    precio_base NUMBER(8,2)
);

-- 21. Proveedores
CREATE TABLE proveedores (
    proveedor_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nombre_proveedor VARCHAR2(100),
    telefono VARCHAR2(20),
    direccion_id NUMBER,
    FOREIGN KEY (direccion_id) REFERENCES direcciones(direccion_id)
);

-- 22. Citas
CREATE TABLE citas (
    cita_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    cliente_id NUMBER NOT NULL,
    vehiculo_id NUMBER NOT NULL,
    sucursal_id NUMBER NOT NULL,
    fecha_cita TIMESTAMP,
    estado VARCHAR2(20) DEFAULT 'AGENDADA',
    FOREIGN KEY (cliente_id) REFERENCES clientes(cliente_id),
    FOREIGN KEY (vehiculo_id) REFERENCES vehiculos(vehiculo_id),
    FOREIGN KEY (sucursal_id) REFERENCES sucursales(sucursal_id)
);

-- 23. Detalles de citas
CREATE TABLE detalles_citas (
    detalle_cita_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    cita_id NUMBER NOT NULL,
    servicio_id NUMBER NOT NULL,
    precio_acordado NUMBER(8,2),
    observaciones VARCHAR2(200),
    FOREIGN KEY (cita_id) REFERENCES citas(cita_id),
    FOREIGN KEY (servicio_id) REFERENCES servicios(servicio_id)
);

-- 24. Asignación de técnicos
CREATE TABLE asignaciones_tecnicos (
    asignacion_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    detalle_cita_id NUMBER,
    personal_id NUMBER,
    porcentaje_participacion NUMBER(5,2),
    FOREIGN KEY (detalle_cita_id) REFERENCES detalles_citas(detalle_cita_id),
    FOREIGN KEY (personal_id) REFERENCES personal(personal_id)
);

-- 25. Historial técnico
CREATE TABLE historial_tecnico (
    historial_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    vehiculo_id NUMBER,
    detalle_cita_id NUMBER,
    descripcion_trabajo CLOB,
    fecha_registro DATE DEFAULT SYSDATE,
    FOREIGN KEY (vehiculo_id) REFERENCES vehiculos(vehiculo_id),
    FOREIGN KEY (detalle_cita_id) REFERENCES detalles_citas(detalle_cita_id)
);

-- 26. Facturas
CREATE TABLE facturas (
    factura_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    numero_factura VARCHAR2(20) NOT NULL UNIQUE,
    cliente_id NUMBER,
    cita_id NUMBER,
    total_factura NUMBER(10,2),
    metodo_pago_id NUMBER,
    fecha_emision DATE DEFAULT SYSDATE,
    FOREIGN KEY (cliente_id) REFERENCES clientes(cliente_id),
    FOREIGN KEY (cita_id) REFERENCES citas(cita_id),
    FOREIGN KEY (metodo_pago_id) REFERENCES metodos_pagos(metodo_pago_id)
);

-- 27. Detalles de factura
CREATE TABLE detalles_facturas (
    detalle_factura_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    factura_id NUMBER,
    tipo_item VARCHAR2(20),
    descripcion_item VARCHAR2(200),
    cantidad NUMBER(8,2),
    total_linea NUMBER(10,2),
    FOREIGN KEY (factura_id) REFERENCES facturas(factura_id)
);


CREATE TABLE historial_pagos (
    pago_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    factura_id NUMBER,
    monto_pagado NUMBER(10,2),
    fecha_pago DATE DEFAULT SYSDATE,
    metodo_pago_id NUMBER,
    observaciones VARCHAR2(200),
    FOREIGN KEY (factura_id) REFERENCES facturas(factura_id),
    FOREIGN KEY (metodo_pago_id) REFERENCES metodos_pagos(metodo_pago_id)
);

CREATE TABLE productos_proveedores (
    producto_id NUMBER,
    proveedor_id NUMBER,
    precio_compra NUMBER(10,2),
    fecha_registro DATE DEFAULT SYSDATE,
    PRIMARY KEY (producto_id, proveedor_id),
    FOREIGN KEY (producto_id) REFERENCES productos(producto_id),
    FOREIGN KEY (proveedor_id) REFERENCES proveedores(proveedor_id)
);
